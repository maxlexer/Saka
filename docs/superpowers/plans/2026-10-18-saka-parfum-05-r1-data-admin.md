# SAKA Parfum R1: Supabase Catalog, USD Only and Admin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace Sanity with a Supabase catalog (20 fixed perfumes, one price per size), remove Riel everywhere, and add a one-password admin for names, availability, prices and paid transactions, with the storefront still working end to end.

**Architecture:** The canonical catalog is `supabase/seed/catalog.json`. A script turns it into a seed migration, and the e2e and local mocks load the same file. Server code reads `products` and `size_prices` through `src/lib/supabase.ts`, and `src/lib/catalog.ts` validates the rows with Zod into a `Catalog`. Pages stay statically cached (`revalidate = 300`), admin saves call `revalidatePath("/", "layout")`, and checkout reads live because it runs in a dynamic POST. Admin auth is a signed httpOnly cookie checked in every admin page and server action, with failed logins rate limited in Supabase.

**Tech Stack:** Next.js 16 App Router (server actions, `useActionState`, `cookies()`, `revalidatePath`), Supabase (`@supabase/supabase-js`, PostgREST), Zod 4, Vitest, Playwright, Node 22.18 type stripping for `.ts` scripts.

**Spec:** `docs/superpowers/specs/2026-10-18-saka-redesign-design.md` (sections 2 to 5, 8, 9). Original spec `docs/superpowers/specs/2026-10-18-saka-parfum-design.md` for everything not changed there. Passes 1 and 2 are complete.

`scripts/lib/catalog-rows.ts`:

```ts
export type CatalogProduct = {
  slug: string;
  name: string;
  family: string;
  description: string;
  notes_top: string[];
  notes_heart: string[];
  notes_base: string[];
  mood: string;
  occasion: string;
  vibes: string[];
  occasions: string[];
  available: boolean;
};

export type CatalogFile = { prices: Record<"10" | "20" | "50", number>; products: CatalogProduct[] };

/** Shared by the seed migration, the e2e seed and the local mock, so the three never drift. */
export function catalogRows({ prices, products }: CatalogFile) {
  return {
    products: products.map((product, i) => ({ ...product, sort: (i + 1) * 10 })),
    prices: Object.entries(prices).map(([size, price]) => ({ size_ml: Number(size), price_usd: price })),
  };
}
```

`scripts/seed-sql.ts`:

```ts
import { readFileSync, writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { catalogRows, type CatalogFile } from "./lib/catalog-rows.ts";

const OUT = "supabase/migrations/20261019000100_catalog_seed.sql";
const text = (value: string) => `'${value.replaceAll("'", "''")}'`;
const list = (values: string[]) => (values.length ? `array[${values.map(text).join(", ")}]` : "'{}'::text[]");

export function seedSql(file: CatalogFile) {
  const { products, prices } = catalogRows(file);
  const rows = products.map((p) =>
    `  (${[text(p.slug), text(p.name), text(p.family), text(p.description), list(p.notes_top), list(p.notes_heart), list(p.notes_base), text(p.mood), text(p.occasion), list(p.vibes), list(p.occasions), p.available, p.sort].join(", ")})`,
  );
  return [
    "-- Generated by scripts/seed-sql.ts from supabase/seed/catalog.json. Do not edit by hand.",
    "insert into products (slug, name, family, description, notes_top, notes_heart, notes_base, mood, occasion, vibes, occasions, available, sort) values",
    rows.join(",\n"),
    "on conflict (slug) do nothing;",
    "",
    "insert into size_prices (size_ml, price_usd) values",
    prices.map((r) => `  (${r.size_ml}, ${r.price_usd})`).join(",\n"),
    "on conflict (size_ml) do nothing;",
    "",
  ].join("\n");
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  writeFileSync(OUT, seedSql(JSON.parse(readFileSync("supabase/seed/catalog.json", "utf8"))));
  console.log(`wrote ${OUT}`);
}
```

```sql
create table products (
  slug text primary key check (slug ~ '^[a-z0-9-]+$'),
  name text not null check (char_length(name) between 1 and 80),
  family text not null check (family in ('fresh', 'woody', 'oriental', 'aromatic')),
  description text not null,
  notes_top text[] not null default '{}',
  notes_heart text[] not null default '{}',
  notes_base text[] not null default '{}',
  mood text not null,
  occasion text not null,
  vibes text[] not null default '{}',
  occasions text[] not null default '{}',
  available boolean not null default true,
  sort integer not null,
  updated_at timestamptz not null default now()
);

create table size_prices (
  size_ml integer primary key check (size_ml in (10, 20, 50)),
  price_usd numeric(8, 2) not null check (price_usd > 0 and price_usd < 1000)
);

create table admin_login_attempts (
  id bigint generated always as identity primary key,
  ip_hash text not null,
  created_at timestamptz not null default now()
);

create index admin_login_attempts_ip_created_idx on admin_login_attempts (ip_hash, created_at);

alter table products enable row level security;
alter table size_prices enable row level security;
alter table admin_login_attempts enable row level security;
```

```js
import { createServer } from "node:http";

const fresh = () => ({ khqr: [], telegram: [] });
export const state = fresh();

// The slice of PostgREST that supabase-js uses for the catalog, prices, login attempts and admin reads.
const tables = { products: [], size_prices: [], admin_login_attempts: [], orders: [] };
const KEYS = { products: "slug", size_prices: "size_ml", admin_login_attempts: "id", orders: "id" };
const RESERVED = new Set(["select", "order", "limit", "offset", "on_conflict", "columns"]);
let nextId = 1;

function matches(row, column, expr) {
  const [op, ...rest] = expr.split(".");
  const value = rest.join(".");
  const cell = row[column];
  if (op === "not") return !matches(row, column, value);
  if (op === "eq") return String(cell) === value;
  if (op === "neq") return String(cell) !== value;
  if (op === "gte") return String(cell) >= value;
  if (op === "lt") return String(cell) < value;
  if (op === "is") return value === "null" ? cell == null : String(cell) === value;
  if (op === "in") return value.slice(1, -1).split(",").map((v) => v.replace(/^"|"$/g, "")).includes(String(cell));
  return true;
}

function postgrest(req, res, url, body) {
  const name = url.pathname.slice("/rest/v1/".length);
  const table = tables[name];
  const prefer = String(req.headers.prefer ?? "");
  const reply = (rows, total = rows.length) => {
    const single = String(req.headers.accept ?? "").includes("vnd.pgrst.object");
    res.writeHead(200, { "content-type": "application/json", "content-range": `0-${Math.max(rows.length - 1, 0)}/${total}` });
    res.end(req.method === "HEAD" ? undefined : JSON.stringify(single ? (rows[0] ?? null) : rows));
  };
  if (!table) {
    res.writeHead(404, { "content-type": "application/json" });
    return res.end(JSON.stringify({ message: `unknown table ${name}` }));
  }
  const filters = [...url.searchParams].filter(([key]) => !RESERVED.has(key));
  const hit = (row) => filters.every(([column, expr]) => matches(row, column, expr));

  if (req.method === "GET" || req.method === "HEAD") {
    let rows = table.filter(hit);
    const order = url.searchParams.get("order");
    if (order) {
      const [column, dir] = order.split(".");
      const sign = dir === "desc" ? -1 : 1;
      rows = [...rows].sort((a, b) => (a[column] > b[column] ? sign : a[column] < b[column] ? -sign : 0));
    }
    const offset = Number(url.searchParams.get("offset") ?? 0);
    const limit = url.searchParams.get("limit");
    return reply(rows.slice(offset, limit ? offset + Number(limit) : undefined), rows.length);
  }
  if (req.method === "POST") {
    const key = KEYS[name];
    const merge = prefer.includes("resolution=merge-duplicates");
    const out = (Array.isArray(body) ? body : [body]).map((row) => {
      const at = row[key] === undefined ? -1 : table.findIndex((r) => r[key] === row[key]);
      if (at >= 0 && merge) return Object.assign(table[at], row);
      const record = { ...(key === "id" ? { id: nextId++ } : {}), created_at: new Date().toISOString(), ...row };
      table.push(record);
      return record;
    });
    return reply(out);
  }
  if (req.method === "PATCH") {
    const rows = table.filter(hit);
    rows.forEach((row) => Object.assign(row, body));
    return reply(rows);
  }
  if (req.method === "DELETE") {
    const keep = table.filter((row) => !hit(row));
    const removed = table.length - keep.length;
    table.splice(0, table.length, ...keep);
    return reply([], removed);
  }
}
```

```ts
export const OUT_OF_STOCK = "aventus";

export async function seedCatalog(url: string, key: string) {
  const headers = { apikey: key, authorization: `Bearer ${key}`, "content-type": "application/json" };
  const call = async (path: string, method: string, body?: unknown, prefer = "") => {
    const res = await fetch(`${url}/rest/v1/${path}`, { method, headers: { ...headers, prefer }, body: body === undefined ? undefined : JSON.stringify(body) });
```

```ts
import { z } from "zod";
import { listPrices, listProducts } from "./supabase";
import { OCCASIONS, SCENT_FAMILIES, SIZES_ML, VIBES, type SizeMl } from "./taxonomy";

const tags = <T extends string>(allowed: readonly T[]) =>
  z.array(z.string()).transform((list) => list.filter((value): value is T => (allowed as readonly string[]).includes(value)));

const productRow = z
  .object({
    slug: z.string().regex(/^[a-z0-9-]+$/),
    name: z.string().min(1),
    family: z.enum(SCENT_FAMILIES),
    description: z.string(),
    notes_top: z.array(z.string()),
    notes_heart: z.array(z.string()),
    notes_base: z.array(z.string()),
    mood: z.string(),
    occasion: z.string(),
    vibes: tags(VIBES),
    occasions: tags(OCCASIONS),
    available: z.boolean(),
    sort: z.number(),
  })
  .transform(({ slug, notes_top, notes_heart, notes_base, ...rest }) => ({
    id: slug,
    ...rest,
    notesTop: notes_top,
    notesHeart: notes_heart,
    notesBase: notes_base,
  }));

const priceRow = z.object({ size_ml: z.union([z.literal(10), z.literal(20), z.literal(50)]), price_usd: z.coerce.number().positive() });

export type Product = z.output<typeof productRow>;
export type SizePrices = Record<SizeMl, number>;
export type Catalog = { products: Product[]; prices: SizePrices };
export type CartCatalog = { names: Record<string, string>; prices: SizePrices };

export const EMPTY_CATALOG: Catalog = { products: [], prices: { 10: 0, 20: 0, 50: 0 } };

export function parseCatalog(rawProducts: unknown[], rawPrices: unknown[]): Catalog {
  const products = rawProducts.flatMap((item) => {
    const parsed = productRow.safeParse(item);
    if (parsed.success) return [parsed.data];
    console.warn(`Skipped product ${(item as { slug?: string } | null)?.slug ?? "unknown"}: ${parsed.error.issues[0]?.message}`);
    return [];
  });
  const prices = Object.fromEntries(z.array(priceRow).parse(rawPrices).map((r) => [r.size_ml, r.price_usd]));
  if (SIZES_ML.some((size) => !(size in prices))) throw new Error("A size price is missing");
  return { products, prices: prices as SizePrices };
}

/**
 * Pages that call this are cached (route `revalidate`) and refreshed by admin saves.
 * Checkout calls it from a dynamic POST, so there it always reads live data.
 */
export async function getCatalog(slugs?: string[]) {
  const [products, prices] = await Promise.all([listProducts(slugs), listPrices()]);
  return parseCatalog(products, prices);
}

export const toCartCatalog = ({ products, prices }: Catalog): CartCatalog => ({
  names: Object.fromEntries(products.map((p) => [p.id, p.name])),
  prices,
});
```

```ts
import { createHash, createHmac, timingSafeEqual } from "node:crypto";
import { cookies } from "next/headers";
import { redirect } from "next/navigation";

export const SESSION_COOKIE = "saka_admin";
export const SESSION_MS = 12 * 60 * 60 * 1000;

const digest = (value: string) => createHash("sha256").update(value).digest();
const sign = (expires: number, secret: string) => createHmac("sha256", secret).update(String(expires)).digest("base64url");

/** Constant time: both sides are hashed to the same length before comparing. */
export function checkPassword(input: string, expected = process.env.ADMIN_PASSWORD ?? "") {
  return expected.length > 0 && timingSafeEqual(digest(input), digest(expected));
}

export function createSession(now: number, secret = process.env.ADMIN_SESSION_SECRET ?? "") {
  if (secret.length < 32) throw new Error("ADMIN_SESSION_SECRET must be at least 32 characters");
  const expires = now + SESSION_MS;
  return `${expires}.${sign(expires, secret)}`;
}

export function verifySession(token: string | undefined, now: number, secret = process.env.ADMIN_SESSION_SECRET ?? "") {
  const [expires, mac] = token?.split(".") ?? [];
  if (!expires || !mac || secret.length < 32 || !(Number(expires) > now)) return false;
  const given = Buffer.from(mac);
  const wanted = Buffer.from(sign(Number(expires), secret));
  return given.length === wanted.length && timingSafeEqual(given, wanted);
}

export async function isAdmin() {
  return verifySession((await cookies()).get(SESSION_COOKIE)?.value, Date.now());
}

/** Call first in every admin page and server action. A layout check alone does not protect actions. */
export async function requireAdmin() {
  if (!(await isAdmin())) redirect("/admin/login");
}
```

```ts
"use server";

import { cookies, headers } from "next/headers";
import { redirect } from "next/navigation";
import { checkPassword, createSession, SESSION_COOKIE, SESSION_MS } from "@/lib/admin-auth";
import { clientIp, hashIp } from "@/lib/ip";
import * as db from "@/lib/supabase";

const LOCK_WINDOW_MS = 15 * 60_000;
const MAX_FAILURES = 5;

export type FormState = { ok: boolean; message: string } | null;

export async function login(_: FormState, form: FormData): Promise<FormState> {
  const ipHash = hashIp(clientIp(await headers()));
  if ((await db.countLoginFailures(ipHash, new Date(Date.now() - LOCK_WINDOW_MS).toISOString())) >= MAX_FAILURES) {
    return { ok: false, message: "Too many attempts. Try again in 15 minutes." };
  }
  if (!checkPassword(String(form.get("password") ?? ""))) {
    await db.recordLoginFailure(ipHash);
    return { ok: false, message: "That password is not right." };
  }
  (await cookies()).set(SESSION_COOKIE, createSession(Date.now()), {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "strict",
    path: "/admin",
    maxAge: SESSION_MS / 1000,
  });
  redirect("/admin");
}

export async function logout() {
  (await cookies()).delete({ name: SESSION_COOKIE, path: "/admin" });
  redirect("/admin/login");
}
```

```ts
  slug: z.string().regex(/^[a-z0-9-]+$/),
  name: z.string().trim().min(1, "Enter a name.").max(80, "Use 80 characters or fewer."),

export async function saveProduct(_: FormState, form: FormData): Promise<FormState> {
  await requireAdmin();
  const input = productInput.safeParse(Object.fromEntries(form));
  if (!input.success) return { ok: false, message: input.error.issues[0].message };
  const { slug, name, available } = input.data;

  }
  revalidatePath("/", "layout");
  return { ok: true, message: "Saved" };
}
```

```ts
const OFFSET_MS = 7 * 60 * 60 * 1000; // Cambodia is UTC+7 all year, with no daylight saving.

export function phnomPenhStart(unit: "day" | "month", now: Date) {
  const local = new Date(now.getTime() + OFFSET_MS);
  return new Date(Date.UTC(local.getUTCFullYear(), local.getUTCMonth(), unit === "day" ? local.getUTCDate() : 1) - OFFSET_MS);
}

export function phnomPenhTime(iso: string | null) {
  const parts = Object.fromEntries(
    new Intl.DateTimeFormat("en-GB", {
      timeZone: "Asia/Phnom_Penh", year: "numeric", month: "2-digit", day: "2-digit",
      hour: "2-digit", minute: "2-digit", hourCycle: "h23",
    })
      .formatToParts(iso ? new Date(iso) : new Date())
      .map((p) => [p.type, p.value]),
  );
  return `${parts.year}-${parts.month}-${parts.day} ${parts.hour}:${parts.minute}`;
}
```

```ts
export type PaidOrder = Pick<OrderRow, "id" | "order_number" | "paid_at" | "customer_name" | "telegram" | "fulfilment" | "map_url" | "items" | "total_usd">;

export async function paidOrders(page: number, perPage: number) {
  const from = (page - 1) * perPage;
  const { data, count, error } = await db()
    .from("orders")
    .select("id, order_number, paid_at, customer_name, telegram, fulfilment, map_url, items, total_usd", { count: "exact" })
    .eq("status", "paid")
    .order("paid_at", { ascending: false })
    .range(from, from + perPage - 1);
  if (error) throw new Error(error.message);
  return { rows: (data ?? []) as PaidOrder[], total: count ?? 0 };
}

export async function paidSince(sinceIso: string) {
  return ok(await db().from("orders").select("total_usd, paid_at").eq("status", "paid").gte("paid_at", sinceIso)) as Pick<OrderRow, "total_usd" | "paid_at">[];
}
```

```tsx
import Link from "next/link";
import { formatUsd } from "@/lib/pricing";
import { paidOrders, paidSince } from "@/lib/supabase";
import { orderNumber } from "@/lib/telegram";
import { phnomPenhStart, phnomPenhTime } from "@/lib/time";

const PER_PAGE = 50;

const sum = (orders: { total_usd: number }[]) => orders.reduce((cents, o) => cents + Math.round(Number(o.total_usd) * 100), 0) / 100;

export default async function TransactionsPage({ searchParams }: { searchParams: Promise<{ page?: string }> }) {
  const page = Math.max(1, Math.floor(Number((await searchParams).page)) || 1);
  const now = new Date();
  const [{ rows, total }, month] = await Promise.all([
    paidOrders(page, PER_PAGE),
    paidSince(phnomPenhStart("month", now).toISOString()),
  ]);
  const dayStart = phnomPenhStart("day", now).getTime();
  const today = month.filter((o) => o.paid_at && Date.parse(o.paid_at) >= dayStart);
  const pages = Math.max(1, Math.ceil(total / PER_PAGE));
  const cards = [{ label: "Today", list: today }, { label: "This month", list: month }];

  return (
    <>
      <h1 className="h2 mt-10">Transactions</h1>
      <dl className="mt-8 grid gap-4 sm:grid-cols-2">
        {cards.map(({ label, list }) => (
          <div key={label} className="rounded-[8px] border border-border bg-surface p-6">
            <dt className="eyebrow">{label}</dt>
            <dd className="h2 mt-3">{formatUsd(sum(list))}</dd>
            <dd className="small mt-1 text-muted">{list.length} paid {list.length === 1 ? "order" : "orders"}</dd>
          </div>
        ))}
      </dl>

      {rows.length === 0 ? (
        <p className="mt-10 text-muted">No paid orders yet.</p>
      ) : (
        <ul className="mt-10 space-y-4">
          {rows.map((o) => (
            <li key={o.id} className="rounded-[8px] border border-border bg-surface p-5">
              <div className="flex flex-wrap items-baseline justify-between gap-2">
                <p className="h3">{orderNumber(o.order_number)}</p>
                <p className="small text-muted">{phnomPenhTime(o.paid_at)}</p>
              </div>
              {o.customer_name ? <p className="mt-2">{o.customer_name}, {o.telegram}</p> : <p className="mt-2 text-muted">Details removed</p>}
              <p className="small mt-1 text-muted">
                {o.fulfilment === "delivery" ? "Delivery" : "Pickup"}
                {o.fulfilment === "delivery" && o.map_url && (
                  <>, <a href={o.map_url} target="_blank" rel="noopener noreferrer" className="link text-text">open map</a></>
                )}
              </p>
              <ul className="small mt-3">
                {o.items.map((i) => (
                  <li key={`${i.productId}-${i.sizeMl}`}>{i.name}, {i.sizeMl}ml x{i.qty}</li>
                ))}
              </ul>
              <p className="mt-3 text-right font-medium">{formatUsd(Number(o.total_usd))}</p>
            </li>
          ))}
        </ul>
      )}

      {pages > 1 && (
        <nav aria-label="Pages" className="mt-8 flex justify-between">
          {page > 1 ? <Link href={`/admin/transactions?page=${page - 1}`} className="link">Newer</Link> : <span />}
          {page < pages && <Link href={`/admin/transactions?page=${page + 1}`} className="link">Older</Link>}
        </nav>
      )}
    </>
  );
}
```
